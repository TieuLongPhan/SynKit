\tikzset{
  iocbadge/.style={
    circle,
    fill=black!80,
    text=white,
    font=\bfseries\scriptsize,
    minimum size=4.6mm,
    inner sep=0.5pt,
    anchor=north west
  },
  iotitle/.style={
    font=\bfseries\footnotesize,
    text=black!85,
    anchor=west,
    inner sep=0pt
  },
  iosubtitle/.style={
    font=\scriptsize,
    text=black!55,
    anchor=north west,
    inner sep=0pt
  },
  iocell/.style={
    rounded corners=3pt,
    draw=black!26,
    fill=black!1,
    line width=0.5pt
  },
  iopanel/.style={
    rounded corners=3pt,
    draw=black!16,

    line width=0.4pt
  },

    rounded corners=2.6pt,
    draw=black!18,
    fill=black!2,
    line width=0.4pt,
    inner sep=5pt
  },
  ioflow/.style={
    -{Latex[length=2.4mm]},

    -{Stealth[length=2.2mm]},

    \node[iosubtitle] at
      ($(synisterTitle.south west)+(0,-0.02)$) {#1};%
  \fi
}
