cmake_minimum_required(VERSION 3.20)
project(synkit_dpo_crn VERSION 1.7.0 LANGUAGES CXX)
find_package(rdkit CONFIG REQUIRED)
find_package(Boost CONFIG REQUIRED)
if(NOT DPO_KERNEL_LIBRARY OR NOT DPO_KERNEL_INCLUDE)
  message(FATAL_ERROR "Pass the verified SynKit DPO kernel library and include directory")
endif()
add_executable(synkit_dpo_crn expand.cpp)
target_compile_features(synkit_dpo_crn PRIVATE cxx_std_20)
target_include_directories(synkit_dpo_crn PRIVATE "${DPO_KERNEL_INCLUDE}")
target_link_libraries(synkit_dpo_crn PRIVATE "${DPO_KERNEL_LIBRARY}"
  RDKit::SmilesParse RDKit::GraphMol RDKit::RDGeneral Boost::headers)
target_compile_definitions(synkit_dpo_crn PRIVATE BOOST_JSON_NO_LIB BOOST_CONTAINER_NO_LIB)
if(NOT MSVC)
  target_compile_options(synkit_dpo_crn PRIVATE -Wall -Wextra)
endif()
