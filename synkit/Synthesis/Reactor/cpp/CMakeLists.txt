cmake_minimum_required(VERSION 3.20)
project(synkit_partial VERSION 1.7.0 LANGUAGES CXX)
add_library(synkit_partial SHARED src/partial.cpp)
target_compile_features(synkit_partial PUBLIC cxx_std_17)
target_compile_definitions(synkit_partial PRIVATE SYNKIT_PARTIAL_BUILD)
target_include_directories(synkit_partial PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
set_target_properties(synkit_partial PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(synkit_partial PRIVATE /W4)
else()
  target_compile_options(synkit_partial PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(SYNKIT_PARTIAL_SANITIZE "Enable address and undefined-behavior checks" OFF)
option(SYNKIT_PARTIAL_STATIC_SANITIZERS "Link GCC sanitizer runtimes statically" OFF)
option(SYNKIT_PARTIAL_BUILD_TESTS "Build standalone C ABI checks" OFF)
if(SYNKIT_PARTIAL_BUILD_TESTS)
  enable_testing()
  add_executable(partial_native_tests tests/partial.cpp)
  target_link_libraries(partial_native_tests PRIVATE synkit_partial)
  add_test(NAME partial_native COMMAND partial_native_tests)
endif()
if(SYNKIT_PARTIAL_SANITIZE AND NOT MSVC)
  enable_testing()
  add_executable(partial_sanitizer_tests src/partial.cpp tests/partial.cpp)
  target_compile_definitions(partial_sanitizer_tests PRIVATE SYNKIT_PARTIAL_BUILD)
  target_include_directories(partial_sanitizer_tests PRIVATE include)
  target_compile_features(partial_sanitizer_tests PRIVATE cxx_std_17)
  target_compile_options(partial_sanitizer_tests PRIVATE
    -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(partial_sanitizer_tests PRIVATE -fsanitize=address,undefined)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND SYNKIT_PARTIAL_STATIC_SANITIZERS)
    target_link_options(partial_sanitizer_tests PRIVATE -static-libasan -static-libubsan)
  endif()
  add_test(NAME partial_sanitizers COMMAND partial_sanitizer_tests)
endif()
option(SYNKIT_PARTIAL_WHEEL "Install into the Python package" OFF)
include(GNUInstallDirs)
if(SYNKIT_PARTIAL_WHEEL)
  set(partial_install synkit/Synthesis/Reactor/_native/lib)
else()
  set(partial_install ${CMAKE_INSTALL_LIBDIR})
endif()
install(TARGETS synkit_partial LIBRARY DESTINATION ${partial_install}
  RUNTIME DESTINATION ${partial_install} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
